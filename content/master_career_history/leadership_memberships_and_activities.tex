\section{Leadership, Memberships, and Activities}

\record{Zygote Press, Cleveland, Ohio}{2011 - 2014}{Member and independent studio artist working in etching, relief, and screenprint. Zygote is a nonprofit community printmaking studio, gallery, archive, and education center that expands artists' access to affordable professional workspace, instruction, exhibitions, residencies, and collaborative development.}
\record{Detroit Color Council}{2012 - 2013}{Member of a nonprofit industry forum promoting communication and education among engineers, designers, scientists, and other professionals involved in automotive color and appearance. The council advances shared technical knowledge and practices intended to improve the quality of colored and decorative materials.}
\record{American Institute of Aeronautics and Astronautics Cincinnati Student Chapter}{2009 - 2011}{Member of a student chapter connecting participants with the aerospace profession and AIAA's emphasis on innovation, technical excellence, and professional collaboration. Participated in Students for the Exploration and Development of Space in 2009, supporting its mission to advance student involvement in space exploration through educational and engineering projects.}

  \item Vice President, 2010.
\end{itemize}

\end{itemize}
